1. 수업 목표

이번 수업에서는 Express 라우터에서 OracleDB를 연결해서 DEPT 테이블 CRUD를 구현합니다.

기능 Method URL 설명
목록 조회 GET /dept 전체 부서 목록 조회
상세 조회 GET /dept/:deptno 특정 부서 1건 조회
등록 화면 GET /dept/register 부서 등록 form 출력
등록 처리 POST /dept/register INSERT 실행
수정 화면 GET /dept/:deptno/modify 기존 데이터 조회 후 수정 form 출력
수정 처리 POST /dept/modify UPDATE 실행
삭제 처리 POST /dept/delete DELETE 실행
핵심 흐름
서버 시작 시 oracledb.createPool()로 커넥션 풀을 만들고, 요청마다 oracledb.getConnection()으로 커넥션을 빌립니다. 사용 후에는 반드시 conn.close()로 풀에 반환합니다.

2. DEPT 테이블

수업용 테이블은 다음과 같이 준비합니다.

CREATE TABLE dept (
    deptno NUMBER PRIMARY KEY,
    dname  VARCHAR2(30),
    loc    VARCHAR2(30)
);

테스트 데이터가 없다면 다음 데이터를 넣습니다.

INSERT INTO dept VALUES (10, 'ACCOUNTING', 'NEW YORK');
INSERT INTO dept VALUES (20, 'RESEARCH', 'DALLAS');
INSERT INTO dept VALUES (30, 'SALES', 'CHICAGO');
INSERT INTO dept VALUES (40, 'OPERATIONS', 'BOSTON');

COMMIT;

4. app.js

서버 시작 파일입니다. 여기서 Express 설정, Pug 설정, OracleDB Pool 생성, 라우터 등록을 처리합니다.

import express from "express";
import path from "path";
import { fileURLToPath } from "url";
import oracledb from "oracledb";
import deptRouter from "./ex3_dept_router.js";

const app = express();
const PORT = 3000;

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

// Oracle 조회 결과를 객체 형태로 받기
oracledb.outFormat = oracledb.OUT_FORMAT_OBJECT;

// Pug 설정
app.set("views", path.join(__dirname, "views"));
app.set("view engine", "pug");

// form 데이터 처리
app.use(express.urlencoded({ extended: true }));
app.use(express.json());

// DB Pool 생성
async function initDB() {
    await oracledb.createPool({
        user: "scott",
        password: "tiger",
        connectString: "localhost:1521/XEPDB1",

        poolMin: 1,
        poolMax: 5,
        poolIncrement: 1,

        queueTimeout: 60000,
        poolTimeout: 60
    });

    console.log("OracleDB Pool 생성 완료");
}

// 기본 주소
app.get("/", (req, res) => {
    res.redirect("/dept");
});

// DEPT 라우터 등록
app.use("/dept", deptRouter);

// 서버 시작
await initDB();

app.listen(PORT, () => {
    console.log(`Server running: http://localhost:${PORT}`);
});

// Ctrl + C 종료 시 Pool 정리
process.on("SIGINT", async () => {
    console.log("\n서버 종료 중...");

    try {
        await oracledb.getPool().close(10);
        process.exit(0);
    } catch (err) {
        console.error(err);
        process.exit(1);
    }
});
설명 포인트
  • oracledb.outFormat = oracledb.OUT_FORMAT_OBJECT는 조회 결과를 객체 배열로 받기 위한 설정입니다.
  • createPool()은 서버 시작 시 한 번만 실행합니다.
  • app.use("/dept", deptRouter)로 DEPT 라우터를 등록합니다.
  • SIGINT는 Ctrl + C로 서버 종료 시 Pool을 정리하기 위한 처리입니다.

5. ex3_dept_router.js

DEPT CRUD 기능을 처리하는 라우터 파일입니다.

주의
/register 라우터는 /:deptno 라우터보다 먼저 작성해야 합니다. 그렇지 않으면 /dept/register 요청을 deptno = "register"로 인식할 수 있습니다. 
// ex3_dept_router.js
import express from "express";
import oracledb from "oracledb";

const router = express.Router();


// 1. 목록 조회
router.get("/", async (req, res, next) => {
    let conn;

    try {
        conn = await oracledb.getConnection();

        const result = await conn.execute(`
            SELECT
                deptno AS "deptno",
                dname  AS "dname",
                loc    AS "loc"
            FROM dept
            ORDER BY deptno
        `);

        res.render("deptList", {
            depts: result.rows
        });

    } catch (e) {
        next(e);

    } finally {
        if (conn) {
            await conn.close();
        }
    }
});


// 2. 등록 화면
router.get("/register", (req, res) => {
    res.render("deptRegister");
});


// 3. 등록 처리
router.post("/register", async (req, res, next) => {
    let conn;

    const { deptno, dname, loc } = req.body;

    try {
        conn = await oracledb.getConnection();

        await conn.execute(
            `
            INSERT INTO dept (
                deptno,
                dname,
                loc
            )
            VALUES (
                :deptno,
                :dname,
                :loc
            )
            `,
            {
                deptno: Number(deptno),
                dname,
                loc
            },
            {
                autoCommit: true
            }
        );

        res.redirect("/dept");

    } catch (e) {
        next(e);

    } finally {
        if (conn) {
            await conn.close();
        }
    }
});


// 4. 수정 화면
router.get("/:deptno/modify", async (req, res, next) => {
    let conn;

    const deptno = Number(req.params.deptno);

    try {
        conn = await oracledb.getConnection();

        const result = await conn.execute(
            `
            SELECT
                deptno AS "deptno",
                dname  AS "dname",
                loc    AS "loc"
            FROM dept
            WHERE deptno = :deptno
            `,
            {
                deptno
            }
        );

        if (result.rows.length === 0) {
            res.redirect("/dept");
            return;
        }

        res.render("deptModify", {
            dept: result.rows[0]
        });

    } catch (e) {
        next(e);

    } finally {
        if (conn) {
            await conn.close();
        }
    }
});


// 5. 수정 처리
router.post("/modify", async (req, res, next) => {
    let conn;

    const { deptno, dname, loc } = req.body;

    try {
        conn = await oracledb.getConnection();

        const result = await conn.execute(
            `
            UPDATE dept
            SET
                dname = :dname,
                loc   = :loc
            WHERE deptno = :deptno
            `,
            {
                deptno: Number(deptno),
                dname,
                loc
            },
            {
                autoCommit: true
            }
        );

        if (result.rowsAffected === 0) {
            res.redirect("/dept");
            return;
        }

        res.redirect(`/dept/${deptno}`);

    } catch (e) {
        next(e);

    } finally {
        if (conn) {
            await conn.close();
        }
    }
});


// 6. 삭제 처리
router.post("/delete", async (req, res, next) => {
    let conn;

    const { deptno } = req.body;

    try {
        conn = await oracledb.getConnection();

        await conn.execute(
            `
            DELETE FROM dept
            WHERE deptno = :deptno
            `,
            {
                deptno: Number(deptno)
            },
            {
                autoCommit: true
            }
        );

        res.redirect("/dept");

    } catch (e) {
        next(e);

    } finally {
        if (conn) {
            await conn.close();
        }
    }
});


// 7. 상세 조회
router.get("/:deptno", async (req, res, next) => {
    let conn;

    const deptno = Number(req.params.deptno);

    try {
        conn = await oracledb.getConnection();

        const result = await conn.execute(
            `
            SELECT
                deptno AS "deptno",
                dname  AS "dname",
                loc    AS "loc"
            FROM dept
            WHERE deptno = :deptno
            `,
            {
                deptno
            }
        );

        if (result.rows.length === 0) {
            res.redirect("/dept");
            return;
        }

        res.render("deptDetail", {
            dept: result.rows[0]
        });

    } catch (e) {
        next(e);

    } finally {
        if (conn) {
            await conn.close();
        }
    }
});


// 8. 에러 미들웨어
router.use((err, req, res, next) => {
    console.error(err);

    res.status(500).send(`
        <h1>서버 오류가 발생했습니다.</h1>
        <p>다시 시도해주세요.</p>
        <a href="/dept">목록으로</a>
    `);
});

export default router;

8. 수업 중 보여주기 좋은 오류

SQL 오타

일부러 SQL을 틀리게 작성해서 오류 처리 흐름을 보여줍니다.

const result = await conn.execute("seleddct * from dept");

실행 흐름:

  1. execute()에서 SQL 오류 발생
  2. catch로 이동
  3. next(e)로 에러 미들웨어 호출
  4. finally에서 커넥션 반환

커넥션 반환 누락

일부러 await conn.close()를 빼고 여러 번 요청을 보내면 커넥션 풀이 부족해질 수 있습니다.

예상 흐름:

  1. 요청마다 커넥션을 빌림
  2. 커넥션을 반환하지 않음
  3. poolMax에 도달
  4. 새 요청이 대기 상태가 됨
  5. queueTimeout 이후 오류 발생 가능

9. 마지막 과제

  1. 등록 시 deptno가 이미 존재하면 오류 메시지 출력하기
  2. 삭제 버튼을 누르면 confirm 창 띄우기
  3. 목록 화면에 검색 기능 추가하기
    예: /dept?keyword=SALES
  4. loc를 input이 아니라 select box로 바꾸기
    예: SEOUL, BUSAN, DALLAS, NEW YORK
  5. 수정 또는 삭제 후 rowsAffected가 0이면 “대상이 존재하지 않습니다” 메시지 출력하기
핵심 정리
Express + OracleDB CRUD의 핵심은 SQL 실행뿐 아니라, 커넥션을 안전하게 빌리고 사용 후 반드시 반환하는 흐름을 만드는 것입니다.
